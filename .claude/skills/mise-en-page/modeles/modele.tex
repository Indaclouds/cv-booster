% \silhouette{width}{height}: neutral placeholder photo (to be replaced by the real one)
\newcommand{\silhouette}[2]{%

    \clip (0,0) rectangle (#1,#2);
    \fill[barrevide] (0,0) rectangle (#1,#2);
    \fill[white] (0.5*#1,0.64*#2) circle (0.19*#1);
    \fill[white] (0.5*#1,0) ellipse [x radius=0.36*#1, y radius=0.38*#2];
  \end{tikzpicture}}

  \vspace{-4pt}}
