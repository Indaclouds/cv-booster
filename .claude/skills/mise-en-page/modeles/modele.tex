% \silhouette{largeur}{hauteur} : photo provisoire neutre (à remplacer par la vraie photo)
\newcommand{\silhouette}[2]{%
  \begin{tikzpicture}
    \clip (0,0) rectangle (#1,#2);
    \fill[barrevide] (0,0) rectangle (#1,#2);
    \fill[white] (0.5*#1,0.64*#2) circle (0.19*#1);
    \fill[white] (0.5*#1,0) ellipse [x radius=0.36*#1, y radius=0.38*#2];
  \end{tikzpicture}}

% \entetephoto{nom}{titre}{contact}{code de l'image}
\newcommand{\entetephoto}[4]{%
  \ifphotogauche
    \begin{minipage}[c]{0.12\linewidth}#4\end{minipage}\hfill

  \else
    \begin{minipage}[c]{0.86\linewidth}\entete{#1}{#2}{#3}\end{minipage}\hfill
    \begin{minipage}[c]{0.12\linewidth}\raggedleft #4\end{minipage}\par
  \fi
  \vspace{-4pt}}
